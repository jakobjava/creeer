const NAILING: u32 = 60;
const PER_BOARD: u32 = 150;
const BOARDS: u32 = 12;
const OVER: u32 = NAILING + BOARDS * PER_BOARD;
/// How far back from a board a nailer stands (m).
const BACK: f64 = 0.75;

/// Where player `k` (Ada north, Ben south) works on board `b`, and the board's crossing with
/// their side's stringer.
fn nailing(play: &Play, site: &BridgeSite, k: usize, b: u32) -> (DVec3, DVec3) {
    let y = if k == 0 { STRINGERS[2] } else { STRINGERS[0] };
    let x = FIRST_BOARD + BOARD_EVERY * f64::from(b);
    let w = &play.server.world;
    (deck(w, site, x - BACK, y), deck(w, site, x, y))
}

/// J at deck board `b`'s crossing.
fn fasten(play: &Play, site: &BridgeSite, b: u32, at: DVec3) -> Command {
    let board = site.item("deck board", b as usize).unwrap_or(EntityId(0));
    #[allow(clippy::cast_precision_loss)]
    let seen = play.server.world.tick as f64;
    Command::Fasten {
        target: Target {
            body: Some(board),
            at,
            seen,
        },
        fastener: "nail_100".into(),
    }
}
